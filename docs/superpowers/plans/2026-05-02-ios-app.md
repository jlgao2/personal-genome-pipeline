# iOS Companion App — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans.

**Goal:** A single-screen iOS app that makes the desktop dashboard's Action Loop usable on the move. Reads HealthKit live (so Garmin daily-rollup data flows through Apple's bridge for free), reads `findings.json` + `cross_refs.json` + `health_profile.json` from iCloud Drive, and writes a daily `samples.json` back so the laptop pipeline can ingest yesterday's vitals without a manual `export.zip` dance.

**Architecture:**

```
┌──────────────────┐  iCloud Drive   ┌────────────────┐
│  laptop pipeline │ ──findings.json─▶  iOS app        │
│  (refresh.sh)    │ ──cross_refs────▶                 │
│                  │ ──profile.json──▶  HKHealthStore  │
│                  │ ◀─samples.json─── (live reads)    │
└──────────────────┘                  └────────────────┘
```

iCloud Drive is the bus — both ends authenticate with Apple ID, no server, no auth ceremony.

**Tech Stack:** Swift 5.10, SwiftUI, HealthKit framework, FileManager + iCloud Drive APIs (`NSMetadataQuery` for change-tracking). No third-party deps in v0. Xcode 15+, iOS 17+ deployment target.

```python
def publish_ios_export(parquet_dir: Path,
                       events_dir: Optional[Path] = None,
                       parquet_root: Optional[Path] = None,
                       profile_path: Optional[Path] = None,
                       outdir: Optional[Path] = None) -> None:
    """Emit JSON files the iOS app reads via iCloud Drive."""
    if events_dir is None:    events_dir = parquet_dir.parent / "events"
    if parquet_root is None:  parquet_root = parquet_dir.parent
    if profile_path is None:  profile_path = Path("output/health_profile.json")
    if outdir is None:        outdir = Path("output/ios_export")
    outdir.mkdir(parents=True, exist_ok=True)
    profile = load_health_profile(profile_path) or {}
    bundle = {
        "exported_at": _dt.datetime.utcnow().isoformat() + "Z",
        "vitals":      build_vitals(parquet_dir),
        "workouts":    build_workouts(events_dir),
        "action_loop": build_action_loop(parquet_root),
        "profile":     profile,
    }
    (outdir / "ios_bundle.json").write_text(json.dumps(bundle, indent=2))
```

- [ ] **Step 2: iCloud bundle loader**

```swift
final class DataLoader {
    static let shared = DataLoader()
    private let containerID = "iCloud.com.example.PersonalData"

    func iCloudURL() -> URL? {
        FileManager.default
            .url(forUbiquityContainerIdentifier: containerID)?
            .appendingPathComponent("Documents/ios_export/ios_bundle.json")
    }

    func loadBundle() async throws -> IOSBundle {
        guard let url = iCloudURL() else { throw DataError.noICloud }
        // Force download if not yet local
        try? FileManager.default.startDownloadingUbiquitousItem(at: url)
        // Wait briefly for sync (simple polling — replace with NSMetadataQuery later)
        for _ in 0..<10 {
            if FileManager.default.fileExists(atPath: url.path) { break }
            try await Task.sleep(nanoseconds: 200_000_000)
        }
        let data = try Data(contentsOf: url)
        return try JSONDecoder().decode(IOSBundle.self, from: data)
    }
}
```

- [ ] **Step 3: HealthKit store**

```swift
import HealthKit

final class HealthStore {
    static let shared = HealthStore()
    let store = HKHealthStore()

    static let readTypes: Set<HKObjectType> = [
        HKObjectType.quantityType(forIdentifier: .restingHeartRate)!,
        HKObjectType.quantityType(forIdentifier: .heartRate)!,
        HKObjectType.quantityType(forIdentifier: .vo2Max)!,
        HKObjectType.quantityType(forIdentifier: .bodyMass)!,
        HKObjectType.quantityType(forIdentifier: .bloodPressureSystolic)!,
        HKObjectType.quantityType(forIdentifier: .bloodPressureDiastolic)!,
        HKObjectType.quantityType(forIdentifier: .stepCount)!,
        HKObjectType.quantityType(forIdentifier: .appleExerciseTime)!,
        HKCategoryType.categoryType(forIdentifier: .sleepAnalysis)!,
    ]

    func authorize() async throws {
        try await store.requestAuthorization(toShare: [], read: HealthStore.readTypes)
    }

    /// Today's resting HR (most recent reading).
    func todayRestingHR() async -> Double? {
        let type = HKQuantityType(.restingHeartRate)
        let pred = HKQuery.predicateForSamples(
            withStart: Calendar.current.startOfDay(for: Date()),
            end: Date(), options: .strictStartDate)
        let descriptor = HKSampleQueryDescriptor(
            predicates: [.quantitySample(type: type, predicate: pred)],
            sortDescriptors: [SortDescriptor(\.endDate, order: .reverse)],
            limit: 1)
        let samples = (try? await descriptor.result(for: store)) ?? []
        return samples.first?.quantity.doubleValue(for: .count().unitDivided(by: .minute()))
    }

    /// Last night's total sleep minutes.
    func lastNightSleepMinutes() async -> Double? {
        let type = HKCategoryType(.sleepAnalysis)
        let cal = Calendar.current
        let startOfYesterday = cal.startOfDay(for: cal.date(byAdding: .day, value: -1, to: Date())!)
        let pred = HKQuery.predicateForSamples(
            withStart: startOfYesterday, end: Date(), options: .strictStartDate)
        let descriptor = HKSampleQueryDescriptor(
            predicates: [.categorySample(type: type, predicate: pred)],
            sortDescriptors: [],
            limit: HKObjectQueryNoLimit)
        let samples = (try? await descriptor.result(for: store)) ?? []
        // Sum all "asleep" buckets (excluding awake)
        let total = samples.reduce(0.0) { acc, s in
            guard s.value != HKCategoryValueSleepAnalysis.awake.rawValue else { return acc }
            return acc + s.endDate.timeIntervalSince(s.startDate)
        }
        return total / 60.0
    }
}
```

- [ ] **Step 1: AppStore (observable state)**

```swift
@MainActor
final class AppStore: ObservableObject {
    @Published var bundle: IOSBundle?
    @Published var liveValues: [String: Double] = [:]   // sample_type → today's reading
    @Published var loading = true

    func bootstrap() async {
        loading = true
        do {
            try await HealthStore.shared.authorize()
            bundle = try await DataLoader.shared.loadBundle()
            await refreshLive()
        } catch {
            print("bootstrap error: \\(error)")
        }
        loading = false
    }

    func refreshLive() async {
        var v: [String: Double] = [:]
        if let rhr = await HealthStore.shared.todayRestingHR() {
            v["heart_rate_resting"] = rhr
        }
        if let s = await HealthStore.shared.lastNightSleepMinutes() {
            v["sleep_minutes"] = s
        }
        // Add weight, BP, VO2Max, etc. as separate queries (one-liner each)
        liveValues = v
    }
}
```

struct ActionLoopView: View {
    let cards: [ActionCard]
    let live: [String: Double]    // overrides latest_value when present

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text("ACTION LOOP")
                .font(.system(.caption2, design: .monospaced))
                .foregroundStyle(.secondary)
            ForEach(cards, id: \\.finding_id) { card in
                let actual = live[card.sample_type] ?? card.latest_value
                ActionCardRow(card: card, actual: actual)
            }
        }
    }
}

struct ActionCardRow: View {
    let card: ActionCard
    let actual: Double?

    var body: some View {
        HStack(alignment: .firstTextBaseline) {
            VStack(alignment: .leading, spacing: 4) {
                Text(card.gene ?? "PRS")
                    .font(.system(.caption2, design: .monospaced))
                    .foregroundStyle(.cyan)
                Text(card.finding_summary ?? card.finding_id)
                    .italic()
                    .font(.body)
                if let take = card.takeaway {
                    Text(take)
                        .font(.footnote)
                        .foregroundStyle(.secondary)
                }
            }
            Spacer()
            VStack(alignment: .trailing) {
                Text(actual.map { String(format: "%.1f", $0) } ?? "—")
                    .font(.system(.title2, design: .serif))
                    .italic()
                    .foregroundStyle(stateColor)
                if let tv = card.target_value {
                    Text("target \\(Int(tv))")
                        .font(.system(.caption2, design: .monospaced))
                        .foregroundStyle(.secondary)
                }
            }
        }
        .padding(.vertical, 8)
        .overlay(alignment: .leading) {
            Rectangle().fill(stateColor).frame(width: 2)
        }
    }

    var stateColor: Color {
        guard let actual = actual, let target = card.target_value else { return .gray }
        let ratio = actual / target
        let dir = card.expected_direction ?? "increase"
        if dir == "increase" {
            if ratio < 0.95 { return .green }
            if ratio < 1.10 { return .cyan }
            return .red
        }
        if ratio >= 1.0  { return .green }
        if ratio >= 0.85 { return .cyan }
        return .red
    }
}

## Task 5 — Daily HealthKit → samples.json upload

**Files:**
- Create: `HealthKit/SampleExporter.swift`

This closes the loop: app reads yesterday's HealthKit data, writes a daily JSON file to iCloud, laptop ingests on next refresh.

- [ ] **Step 1: Build the exporter**

```swift
struct SampleExporter {
    static func dailySamples(for day: Date) async -> [[String: Any]] {
        var rows: [[String: Any]] = []
        let cal = Calendar.current
        let dayStart = cal.startOfDay(for: day)
        let dayEnd   = cal.date(byAdding: .day, value: 1, to: dayStart)!

        // Resting HR — single value per day
        if let rhr = await HealthStore.shared.todayRestingHR() {
            rows.append([
                "ts": dayStart.iso8601,
                "ts_end": dayStart.iso8601,
                "source": "healthkit",
                "type": "heart_rate_resting",
                "value": rhr,
                "unit": "bpm",
                "meta": "{}",
            ])
        }
        if let sleep = await HealthStore.shared.lastNightSleepMinutes() {
            rows.append([
                "ts": dayStart.iso8601,
                "ts_end": dayStart.iso8601,
                "source": "healthkit",
                "type": "sleep_minutes",
                "value": sleep,
                "unit": "min",
                "meta": "{}",
            ])
        }
        // ... weight, BP, etc.
        return rows
    }

    static func write(rows: [[String: Any]]) throws {
        guard let dir = FileManager.default
            .url(forUbiquityContainerIdentifier: "iCloud.com.example.PersonalData")?
            .appendingPathComponent("Documents/ios_export") else { return }
        try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        let url = dir.appendingPathComponent("samples_\\(Date().iso8601date).json")
        let data = try JSONSerialization.data(withJSONObject: rows, options: .prettyPrinted)
        try data.write(to: url)
    }
}
```

## Acceptance criteria

- [ ] iCloud sync works: laptop writes `ios_bundle.json`, app reads it within 30 seconds.
- [ ] HealthKit auth dialog appears once on first launch; data flows after.
- [ ] Action Loop renders with at least 5 cards (gray for missing data, colored for live).

## What this plan does NOT do

- **Direct API access to Garmin, MyChart, etc.** — all upstream data flows through HealthKit (which already has bridges from Garmin Connect and MyChart Health Records).
- **Push notifications.** Future: notify when a vital crosses a threshold.
- **Social sharing.** Single-user, no accounts.
- **Apple Watch app for tap-to-log supplements.** Future enhancement.
